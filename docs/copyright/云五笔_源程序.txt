  677      // ===== v0.4.8 数字面板四则计算（先乘除后加减，左结合） =====

  679      private static double calcEval(String expr) {
  680          expr = expr.replace('×', '*').replace('÷', '/');
  681          List<Double> nums = new ArrayList<>();
  682          List<Character> ops = new ArrayList<>();
  683          StringBuilder cur = new StringBuilder();
  684          for (int i = 0; i < expr.length(); i++) {
  685              char c = expr.charAt(i);
  686              if (c == '+' || c == '-' || c == '*' || c == '/') {
  687                  if (cur.length() == 0) return Double.NaN;
  688                  nums.add(Double.parseDouble(cur.toString()));
  689                  cur.setLength(0);
  690                  ops.add(c);
  691              } else if (c >= '0' && c <= '9' || c == '.') {
  692                  cur.append(c);
  693              } else {
  694                  return Double.NaN;
  695              }
  696          }
  697          if (cur.length() == 0) return Double.NaN;
  698          nums.add(Double.parseDouble(cur.toString()));
  699          // 先乘除
  700          for (int i = 0; i < ops.size(); i++) {

第15页
  701              char op = ops.get(i);
  702              if (op == '*' || op == '/') {
  703                  double a = nums.get(i), b = nums.get(i + 1);
  704                  double r = (op == '*') ? a * b : (b == 0 ? Double.NaN : a / b);
  705                  if (Double.isNaN(r)) return Double.NaN;
  706                  nums.set(i, r);
  707                  nums.remove(i + 1);
  708                  ops.remove(i);
  709                  i--;
  710              }
  711          }
  712          // 再加减（左结合）
  713          double acc = nums.get(0);
  714          for (int i = 0; i < ops.size(); i++) {
  715              char op = ops.get(i);
  716              double b = nums.get(i + 1);
  717              acc = (op == '+') ? acc + b : acc - b;
  718          }
  719          return acc;
  720      }

 1482          int c = key.codes[0];

 1486      private Keyboard.Key findKey(int x, int y) {
 1487          Keyboard kb = getKeyboard();
 1488          if (kb == null) return null;
 1489          // v0.5.46 反馈①：触摸坐标含 padding，键坐标不含——先减 padding 再命中（修复"打 r 打到 t"键位错）
 1490          x -= getPaddingLeft();
 1491          y -= getPaddingTop();
 1492          for (Keyboard.Key key : kb.getKeys()) {
 1493              if (x >= key.x && x <= key.x + key.width
 1494                      && y >= key.y && y <= key.y + key.height) {
 1495                  return key;
 1496              }
 1497          }
 1498          return null;
